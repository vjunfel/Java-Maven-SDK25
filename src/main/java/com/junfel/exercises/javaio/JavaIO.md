Java I/O Hierarchy

Object
 ├── InputStream
 │     ├── FileInputStream
 │     ├── BufferedInputStream
 │     └── ObjectInputStream
 │
 ├── OutputStream
 │     ├── FileOutputStream
 │     ├── BufferedOutputStream
 │     └── ObjectOutputStream
 │
 ├── Reader
 │     ├── FileReader
 │     ├── BufferedReader
 │     └── InputStreamReader
 │
 └── Writer
       ├── FileWriter
       ├── BufferedWriter
       └── PrintWriter
